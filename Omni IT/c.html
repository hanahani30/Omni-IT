البرنامج الأول - Hello World

#include <stdio.h>

int main() {
    printf("مرحباً في C!\n");
    return 0;
}

النتيجة:
مرحباً في C!

شرح الكود:
├─ #include <stdio.h> : تضمين مكتبة الإدخال والإخراج
├─ int main() : دالة البرنامج الرئيسية
├─ printf() : طباعة النص
└─ return 0 : إنهاء البرنامج بنجاح

المتغيرات والأنواع

#include <stdio.h>

int main() {
    // الأنواع الأساسية
    int age = 25;              // عدد صحيح
    float height = 1.75;       // عدد عشري
    double precise = 3.14159;  // دقة أعلى
    char initial = 'A';        // حرف واحد
    
    printf("العمر: %d\n", age);
    printf("الطول: %.2f\n", height);
    printf("القيمة: %.5f\n", precise);
    printf("الحرف: %c\n", initial);
    
    // المصفوفات
    int numbers[5] = {1, 2, 3, 4, 5};
    
    printf("الأرقام: ");
    for (int i = 0; i < 5; i++) {
        printf("%d ", numbers[i]);
    }
    printf("\n");
    
    return 0;
}

النتيجة:
العمر: 25
الطول: 1.75
القيمة: 3.14159
الحرف: A
الأرقام: 1 2 3 4 5

المؤشرات والذاكرة

#include <stdio.h>
#include <stdlib.h>

int main() {
    // المؤشر (Pointer)
    int x = 10;
    int *ptr = &x;  // مؤشر يشير إلى x
    
    printf("قيمة x: %d\n", x);
    printf("عنوان x: %p\n", (void*)&x);
    printf("قيمة المؤشر: %p\n", (void*)ptr);
    printf("قيمة من خلال المؤشر: %d\n", *ptr);
    
    // تخصيص ذاكرة ديناميكية
    int *array = (int*)malloc(5 * sizeof(int));
    
    for (int i = 0; i < 5; i++) {
        array[i] = (i + 1) * 10;
    }
    
    printf("المصفوفة المخصصة: ");
    for (int i = 0; i < 5; i++) {
        printf("%d ", array[i]);
    }
    printf("\n");
    
    // تحرير الذاكرة
    free(array);
    
    return 0;
}

النتيجة:
قيمة x: 10
قيمة المؤشر: 0x7fff...
قيمة من خلال المؤشر: 10
المصفوفة المخصصة: 10 20 30 40 50
