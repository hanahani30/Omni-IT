fun main() {
    // متغير ثابت
    val name = "أحمد"
    
    // متغير قابل للتغيير
    var age = 25
    age = 26
    
    // تحديد النوع صراحة
    val email: String = "ahmed@example.com"
    val score: Int = 95
    val height: Double = 1.75
    
    println("الاسم: $name")
    println("العمر: $age")
    println("البريد: $email")
    println("الطول: $height متر")
    
    // المصفوفات
    val numbers = listOf(1, 2, 3, 4, 5)
    val students = listOf("أحمد", "فاطمة", "علي")
    
    println("الأرقام: $numbers")
    println("الطلاب: $students")
}

الفئات والكائنات

// فئة بسيطة
class Student(
    val name: String,
    val age: Int,
    val gpa: Double
) {
    fun display() {
        println("الاسم: $name")
        println("العمر: $age")
        println("المعدل: $gpa")
    }
}

// فئة مشتقة
class GraduateStudent(
    name: String,
    age: Int,
    gpa: Double,
    val graduationYear: Int
) : Student(name, age, gpa) {
    fun displayGraduation() {
        println("سنة التخرج: $graduationYear")
    }
}

fun main() {
    val student = Student("فاطمة", 20, 3.8)
    student.display()
    
    val grad = GraduateStudent("علي", 25, 3.9, 2023)
    grad.display()
    grad.displayGraduation()
}

النتيجة:
الاسم: فاطمة
العمر: 20
المعدل: 3.8
الاسم: علي
العمر: 25
المعدل: 3.9
سنة التخرج: 2023
