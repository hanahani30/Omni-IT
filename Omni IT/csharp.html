الفئات والكائنات

using System;

// تعريف فئة Student
class Student {
    public string Name { get; set; }
    public int Age { get; set; }
    public double GPA { get; set; }
    
    // بناء الفئة
    public Student(string name, int age, double gpa) {
        Name = name;
        Age = age;
        GPA = gpa;
    }
    
    // دالة لعرض المعلومات
    public void Display() {
        Console.WriteLine($"الاسم: {Name}");
        Console.WriteLine($"العمر: {Age}");
        Console.WriteLine($"المعدل: {GPA}");
    }
}

class Program {
    static void Main() {
        // إنشاء كائن
        Student student = new Student("أحمد", 20, 3.8);
        student.Display();
        
        // كائن آخر
        Student student2 = new Student("فاطمة", 21, 3.9);
        student2.Display();
    }
}

النتيجة:
الاسم: أحمد
العمر: 20
المعدل: 3.8
الاسم: فاطمة
العمر: 21
المعدل: 3.9

الواجهات والوراثة

using System;

// واجهة (Interface)
interface IAnimal {
    void Speak();
    void Move();
}

// فئة قاعدية
class Animal {
    public string Name { get; set; }
    
    public Animal(string name) {
        Name = name;
    }
}

// فئة مشتقة
class Dog : Animal, IAnimal {
    public Dog(string name) : base(name) { }
    
    public void Speak() {
        Console.WriteLine($"{Name} يعوي: وووووو!");
    }
    
    public void Move() {
        Console.WriteLine($"{Name} يركض بسرعة");
    }
}

class Program {
    static void Main() {
        Dog dog = new Dog("ماكس");
        dog.Speak();
        dog.Move();
    }
}

النتيجة:
ماكس يعوي: وووووو!
ماكس يركض بسرعة
