الفئات والهياكل

import Foundation

// الهيكل (Struct)
struct Person {
    var name: String
    var age: Int
    
    func describe() -> String {
        return "\(name) عمره \(age) سنة"
    }
}

// الفئة (Class)
class Student {
    var name: String
    var age: Int
    var gpa: Double
    
    init(name: String, age: Int, gpa: Double) {
        self.name = name
        self.age = age
        self.gpa = gpa
    }
    
    func display() {
        print("الاسم: \(name)")
        print("العمر: \(age)")
        print("المعدل: \(gpa)")
    }
}

let person = Person(name: "أحمد", age: 25)
print(person.describe())

let student = Student(name: "فاطمة", age: 20, gpa: 3.8)
student.display()

النتيجة:
أحمد عمره 25 سنة
الاسم: فاطمة
العمر: 20
المعدل: 3.8

📱 استخدامات Swift الواقعية

1. تطبيقات iOS
   └─ مثال: تطبيقات الهواتف الذكية
      ├─ تطبيقات Apple الرسمية
      ├─ Snapchat (محرك الكاميرا)
      ├─ Twitter (جزء من التطبيق)
      └─ ملايين التطبيقات

2. تطبيقات macOS
   └─ مثال: برامج Mac
      ├─ Xcode (محرر Apple)
      ├─ برامج الإنتاجية
      ├─ برامج الألعاب
      └─ برامج احترافية

3. تطبيقات watchOS
   └─ مثال: Apple Watch
      ├─ تطبيقات الساعة الذكية
      ├─ تأثيرات الصيانة
      ├─ تطبيقات اللياقة
      └─ إشعارات ذكية

4. خوادم الويب
   └─ مثال: Vapor framework
      ├─ APIs RESTful
      ├─ تطبيقات الويب الكاملة
      └─ خوادم سريعة وفعالة
