حماية تطبيقات الويب

البرنامج الأول - Hello World

<?php
    echo "مرحباً في PHP!";
?>

النتيجة على الصفحة:
مرحباً في PHP!

شرح الكود:
├─ <?php و ?> : بداية ونهاية كود PHP
└─ echo : طباعة النص على الصفحة

المتغيرات والأنواع

<?php
    // المتغيرات تبدأ بـ $
    $name = "أحمد";
    $age = 25;
    $height = 1.75;
    $is_student = true;
    
    echo "الاسم: " . $name . "<br>";
    echo "العمر: " . $age . "<br>";
    echo "الطول: " . $height . " متر<br>";
    echo "هل هو طالب: " . ($is_student ? "نعم" : "لا") . "<br>";
?>

النتيجة:
الاسم: أحمد
العمر: 25
الطول: 1.75 متر
هل هو طالب: نعم

الدوال والعمليات

<?php
    // دالة لحساب المجموع
    function sum($a, $b) {
        return $a + $b;
    }
    
    $result = sum(10, 20);
    echo "مجموع 10 و 20 = " . $result . "<br>";
    
    // مثال بـ array (مصفوفة)
    $students = array("أحمد", "فاطمة", "علي");
    
    echo "عدد الطلاب: " . count($students) . "<br>";
    
    // حلقة for
    for ($i = 0; $i < count($students); $i++) {
        echo "الطالب " . ($i + 1) . ": " . $students[$i] . "<br>";
    }
?>

النتيجة:
مجموع 10 و 20 = 30
عدد الطلاب: 3
الطالب 1: أحمد
الطالب 2: فاطمة
الطالب 3: علي

الاتصال بقاعدة البيانات

<?php
    // الاتصال بقاعدة البيانات MySQL
    $servername = "localhost";
    $username = "root";
    $password = "";
    $database = "school";
    
    // إنشاء الاتصال
    $conn = new mysqli($servername, $username, $password, $database);
    
    // التحقق من الاتصال
    if ($conn->connect_error) {
        die("خطأ في الاتصال: " . $conn->connect_error);
    }
    
    echo "تم الاتصال بنجاح!<br>";
    
    // استعلام بسيط
    $sql = "SELECT * FROM students";
    $result = $conn->query($sql);
    
    if ($result->num_rows > 0) {
        while ($row = $result->fetch_assoc()) {
            echo "الاسم: " . $row["name"] . " - العمر: " . $row["age"] . "<br>";
        }
    }
    
    $conn->close();
?>

🌐 استخدامات PHP الواقعية

1. المتاجر الإلكترونية
   └─ مثال: متجر الإلكترونيات
      ├─ عرض المنتجات من قاعدة البيانات
      ├─ عربة التسوق
      ├─ معالجة الدفع
      └─ قائمة الطلبات

2. إنظمة إدارة المحتوى
   └─ مثال: WordPress
      ├─ إضافة ومعالجة المقالات
      ├─ التعليقات
      ├─ الوسوم والفئات
      └─ حقوق المستخدمين

3. مواقع التواصل الاجتماعي
   └─ مثال: Facebook (في الأصل مبني بـ PHP)
      ├─ حفظ بيانات المستخدمين
      ├─ المنشورات والتعليقات
      ├─ نظام الإشعارات
      └─ نظام الأصدقاء

4. لوحات التحكم الإدارية
   └─ مثال: لوحة تحكم المدرسة
      ├─ إدارة الطلاب
      ├─ إدارة المعلمين
      ├─ نظام العلامات
      └─ التقارير المختلفة
