المفاهيم الأساسية في C++

البرنامج الأول - Hello World:
═════════════════════════════
#include <iostream>
using namespace std;

int main() {
    cout << "مرحباً في C++" << endl;
    return 0;
}

شرح الأسطر:
├─ #include <iostream> : تضمين مكتبة الإدخال والإخراج
├─ using namespace std : استخدام المجال المعياري
├─ int main() { ... } : دالة البرنامج الرئيسية
├─ cout : طباعة النص على الشاشة
└─ return 0 : إنهاء البرنامج

البرمجة الكائنية (OOP)

مثال عملي - إنشاء فئة Student:
═════════════════════════════

#include <iostream>
using namespace std;

class Student {
private:
    string name;
    int age;
    double gpa;

public:
    // دالة البناء (Constructor)
    Student(string n, int a, double g) {
        name = n;
        age = a;
        gpa = g;
    }
    
    // دالة لعرض البيانات
    void display() {
        cout << "الاسم: " << name << endl;
        cout << "العمر: " << age << endl;
        cout << "المعدل: " << gpa << endl;
    }
};

int main() {
    // إنشاء كائن
    Student student1("أحمد", 20, 3.8);
    student1.display();
    
    // إنشاء كائن آخر
    Student student2("فاطمة", 21, 3.9);
    student2.display();
    
    return 0;
}

النتيجة:
الاسم: أحمد
العمر: 20
المعدل: 3.8
الاسم: فاطمة
العمر: 21
المعدل: 3.9

الحلقات والشرط

حلقة for:
═════════════════════════════
for (int i = 1; i <= 5; i++) {
    cout << i << " ";
}
النتيجة: 1 2 3 4 5

حلقة while:
═════════════════════════════
int count = 0;
while (count < 3) {
    cout << "مرحباً " << count << endl;
    count++;
}
النتيجة:
مرحباً 0
مرحباً 1
مرحباً 2

شرط if-else:
═════════════════════════════
int age = 18;

if (age < 13) {
    cout << "أنت طفل" << endl;
} 
else if (age < 18) {
    cout << "أنت مراهق" << endl;
} 
else {
    cout << "أنت بالغ" << endl;
}
النتيجة: أنت بالغ
