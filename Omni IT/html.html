🌍 HTML

لغة لإنشاء هياكل صفحات الويب

مقدمة عن HTML

HTML (HyperText Markup Language) هي لغة لإنشاء صفحات الويب، تُستخدم لإنشاء الهيكل والمحتوى. تعتمد على الوسوم (Tags) لتعريف عناصر الصفحة.

الهيكل الأساسي لصفحة HTML

<!DOCTYPE html>
<html lang="ar">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width">
    <title>عنوان الصفحة</title>
</head>
<body>
    <h1>مرحباً</h1>
    <p>هذا نص في الصفحة</p>
</body>
</html>

العناوين والفقرات

<!-- عناوين (Headings) -->
<h1>العنوان الرئيسي</h1>
<h2>عنوان ثانوي</h2>
<h3>عنوان أصغر</h3>

<!-- فقرات (Paragraphs) -->
<p>هذا نص العنوان الأول</p>
<p>هذا نص العنوان الثاني</p>

الروابط والصور

<!-- الروابط (Links) -->
<a href="https://google.com">اضغط هنا</a>
<a href="page2.html">انتقل للصفحة الثانية</a>

<!-- الصور (Images) -->
<img src="image.jpg" alt="وصف الصورة">
<img src="logo.png" width="200" height="100">

القوائم (Lists)

قائمة غير مرتبة:

<ul>
    <li>العنصر الأول</li>
    <li>العنصر الثاني</li>
    <li>العنصر الثالث</li>
</ul>

قائمة مرتبة:

<ol>
    <li>الخطوة الأولى</li>
    <li>الخطوة الثانية</li>
    <li>الخطوة الثالثة</li>
</ol>

النماذج (Forms)

<form>
    <label for="name">الاسم:</label>
    <input type="text" id="name" name="name">

    <label for="email">البريد الإلكتروني:</label>
    <input type="email" id="email" name="email">

    <label for="message">الرسالة:</label>
    <textarea id="message" name="message"></textarea>

    <button type="submit">إرسال</button>
</form>

الوسوم الدلالية (Semantic Tags)

<header>
    <h1>رأس الصفحة</h1>
</header>

<nav>
    <a href="#">رابط 1</a>
    <a href="#">رابط 2</a>
</nav>

<main>
    <article>
        <h2>عنوان المقال</h2>
        <p>محتوى المقال...</p>
    </article>
</main>

<footer>
    <p>جميع الحقوق محفوظة</p>
</footer>
