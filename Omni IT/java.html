مقدمة عن Java

Java هي لغة برمجة قوية وموثوقة، تعمل على جميع الأنظمة والمنصات. تُستخدم في تطوير التطبيقات الكبيرة والنظم المؤسسية حول العالم.

البنية الأساسية

public class HelloWorld {
    public static void main(String[] args) {
        System.out.println("مرحباً في Java!");
    }
}

الناتج:

مرحباً في Java!

المتغيرات وأنواع البيانات

// أنواع البيانات الأساسية
int age = 25;                    // عدد صحيح
double height = 1.75;            // عدد عشري
String name = "أحمد";            // نص
boolean isStudent = true;        // قيمة منطقية
char letter = 'A';              // حرف واحد

// طباعة المتغيرات
System.out.println(name);
System.out.println(age);
System.out.println("العمر: " + age);

العمليات الحسابية

int a = 10;
int b = 3;

// العمليات الحسابية
System.out.println(a + b);   // 13
System.out.println(a - b);   // 7
System.out.println(a * b);   // 30
System.out.println(a / b);   // 3
System.out.println(a % b);   // 1

// المقارنة
System.out.println(a > b);   // true
System.out.println(a == b);  // false
System.out.println(a != b);  // true

الشروط (if/else)

int age = 18;

if (age >= 18) {
    System.out.println("أنت بالغ");
} else if (age >= 13) {
    System.out.println("أنت مراهق");
} else {
    System.out.println("أنت طفل");
}

الحلقات (Loops)

حلقة for:

// حلقة بسيطة
for (int i = 0; i < 5; i++) {
    System.out.println(i);  // 0, 1, 2, 3, 4
}

// حلقة على مصفوفة
String[] fruits = {"تفاح", "موز", "برتقال"};
for (String fruit : fruits) {
    System.out.println(fruit);
}

حلقة while:

int count = 0;
while (count < 5) {
    System.out.println(count);
    count++;
}

الدوال (Methods)

public class Calculator {
    // دالة لجمع رقمين
    public static int add(int a, int b) {
        return a + b;
    }
    
    // دالة لطباعة رسالة
    public static void greet(String name) {
        System.out.println("مرحباً " + name);
    }
    
    public static void main(String[] args) {
        System.out.println(add(5, 3));
        greet("أحمد");
    }
}

الكلاسات والكائنات (Classes & Objects)

public class Person {
    String name;
    int age;
    
    // Constructor
    public Person(String name, int age) {
        this.name = name;
        this.age = age;
    }
    
    // Method
    public void introduce() {
        System.out.println("أنا " + name + " وعمري " + age);
    }
}

// الاستخدام
Person person = new Person("محمد", 25);
person.introduce();  // أنا محمد وعمري 25
