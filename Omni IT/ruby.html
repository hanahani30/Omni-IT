الدوال والفئات

# دالة بسيطة
def greet(name)
    "مرحباً يا #{name}!"
end

puts greet("أحمد")

# فئة
class Student
    attr_accessor :name, :age, :gpa
    
    def initialize(name, age, gpa)
        @name = name
        @age = age
        @gpa = gpa
    end
    
    def display
        puts "الاسم: #{@name}"
        puts "العمر: #{@age}"
        puts "المعدل: #{@gpa}"
    end
end

# إنشاء كائن
student = Student.new("فاطمة", 20, 3.8)
student.display

# الحلقات
[1, 2, 3, 4, 5].each do |number|
    puts "الرقم: #{number}"
end

النتيجة:
مرحباً يا أحمد!
الاسم: فاطمة
العمر: 20
المعدل: 3.8
الرقم: 1
