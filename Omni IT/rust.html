المتغيرات والملكية

نظام الملكية في Rust:
═════════════════════════════

fn main() {
    // المتغيرات في Rust غير قابلة للتغيير بشكل افتراضي
    let x = 5;
    // x = 6; // خطأ! لا يمكن تغييرها
    
    // لتجعلها قابلة للتغيير
    let mut y = 5;
    y = 6; // صحيح!
    println!("y = {}", y);
    
    // نظام الملكية
    let s1 = String::from("مرحباً");
    let s2 = s1; // الملكية انتقلت من s1 إلى s2
    
    // println!("{}", s1); // خطأ! s1 لا يملك السلسلة بعد الآن
    println!("{}", s2); // صحيح!
}

نظام الاستعارة (Borrowing):
═════════════════════════════

fn main() {
    let s1 = String::from("مرحباً");
    
    // استعارة immutable
    let s2 = &s1;
    println!("{}", s2);
    println!("{}", s1); // لا يزال بإمكاني استخدام s1
    
    // استعارة mutable
    let mut s3 = String::from("مرحباً");
    borrow_mut(&mut s3);
    println!("{}", s3);
}

fn borrow_mut(s: &mut String) {
    s.push_str(" يا صديقي");
}

معالجة الأخطاء

نوع Result في Rust:
═════════════════════════════

fn divide(a: i32, b: i32) -> Result<i32, String> {
    if b == 0 {
        Err(String::from("لا يمكن القسمة على صفر"))
    } else {
        Ok(a / b)
    }
}

fn main() {
    match divide(10, 2) {
        Ok(result) => println!("النتيجة: {}", result),
        Err(e) => println!("خطأ: {}", e),
    }
    
    match divide(10, 0) {
        Ok(result) => println!("النتيجة: {}", result),
        Err(e) => println!("خطأ: {}", e),
    }
}

النتيجة:
النتيجة: 5
خطأ: لا يمكن القسمة على صفر

🚀 استخدامات Rust الواقعية

1. أنظمة التشغيل والبرامج الموثوقة
   └─ مثال: Linux Kernel (جزء منه)
      ├─ دعم محركات الأقراص الجديدة
      ├─ سائقي الأجهزة
      └─ أمان عالي جداً

2. متصفحات الويب
   └─ مثال: Firefox و Servo
      ├─ محرك التصيير
      ├─ معالجات الصور
      └─ كفاءة عالية جداً

3. تطبيقات الحوسبة السحابية
   └─ مثال: Cloudflare Workers
      ├─ معالجة ملايين الطلبات
      ├─ استهلاك منخفض للموارد
      └─ بدء سريع جداً

4. البرامج المدمجة
   └─ مثال: الذكاء الاصطناعي على الأجهزة
      ├─ معالجات خفيفة الوزن
      ├─ استهلاك طاقة منخفض
      └─ أداء عالي
