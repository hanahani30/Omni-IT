لغة برمجة قوية للويب التفاعلي

مقدمة عن JavaScript

JavaScript هي لغة برمجية تعمل في المتصفح والخادم، تُستخدم لإضافة التفاعل والحركة لصفحات الويب. وهي اللغة الأساسية لتطوير تطبيقات ويب حديثة وتفاعلية.

طباعة على Console

لطباعة الرسائل في وحدة التحكم (Console):

console.log("مرحباً في JavaScript!");
console.log(123);
console.log("النتيجة:", 5 + 3);

المتغيرات (Variables)

المتغيرات لتخزين البيانات والقيم:

// استخدام let و const (الحديثة)
let name = "أحمد";
const age = 25;
var height = 1.75;  // قديمة

// تغيير القيمة
name = "محمد";

// طباعة المتغيرات
console.log(name);
console.log(age);
console.log(height);

أنواع البيانات

// String (نص)
let text = "Hello World";

// Number (عدد)
let number = 42;
let decimal = 3.14;

// Boolean (صح/خطأ)
let flag = true;

// Array (مصفوفة/قائمة)
let fruits = ["تفاح", "موز", "برتقال"];
let numbers = [1, 2, 3, 4, 5];

// Object (كائن)
let person = {
    name: "أحمد",
    age: 25,
    city: "الرياض"
};

العمليات الحسابية والمنطقية

// حسابية
console.log(10 + 5);    // 15
console.log(10 - 5);    // 5
console.log(10 * 5);    // 50
console.log(10 / 5);    // 2
console.log(10 % 3);    // 1

// مقارنة
console.log(10 > 5);    // true
console.log(10 == 10);  // true
console.log(10 === "10"); // false

// منطقية
console.log(true && false);  // false
console.log(true || false);  // true
console.log(!true);          // false

الشروط (if/else)

let age = 18;

if (age >= 18) {
    console.log("أنت بالغ");
} else if (age >= 13) {
    console.log("أنت مراهق");
} else {
    console.log("أنت طفل");
}

الحلقات (Loops)

حلقة for:

for (let i = 0; i < 5; i++) {
    console.log(i);  // 0, 1, 2, 3, 4
}

حلقة while:

let count = 0;
while (count < 5) {
    console.log(count);
    count++;
}

forEach:

let fruits = ["تفاح", "موز", "برتقال"];
fruits.forEach(function(fruit) {
    console.log(fruit);
});

الدوال (Functions)

// دالة بسيطة
function greet(name) {
    return "مرحباً " + name;
}
console.log(greet("أحمد"));

// Arrow Function (حديثة)
const add = (a, b) => {
    return a + b;
};
console.log(add(5, 3));  // 8

// Arrow Function مختصرة
const multiply = (a, b) => a * b;
console.log(multiply(4, 5));  // 20

التعامل مع الأحداث (Events)

// HTML
// <button id="myBtn">اضغط هنا</button>

// JavaScript
let button = document.getElementById("myBtn");

button.addEventListener("click", function() {
    console.log("تم الضغط على الزر!");
});

// أو باستخدام Arrow Function
button.addEventListener("click", () => {
    alert("مرحباً!");
});
